{# A signed change. higher_is_better picks the colour; "since" is read out to screen readers. #} {% macro delta(value, since="", higher_is_better=true, decimals=1) -%} {%- if value is none -%} — {%- else -%} {%- set shown = ("%." ~ decimals ~ "f")|format(value|abs) if decimals else "{:,}".format(value|abs) -%} {%- if value == 0 -%} no change{{ " since " ~ since if since }} {%- else -%} {%- set good = (value > 0) == higher_is_better -%} {{ "▲" if value > 0 else "▼" }} {{ shown }} {{ "up" if value > 0 else "down" }}{{ " since " ~ since if since }} {%- endif -%} {%- endif -%} {%- endmacro %}
